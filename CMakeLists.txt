cmake_minimum_required(VERSION 3.16)
project(OfficialCraftCpp VERSION 3.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
add_executable(officialcraft src/main.cpp)
target_link_libraries(officialcraft PRIVATE Qt6::Widgets Qt6::Network)
target_compile_options(officialcraft PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS officialcraft RUNTIME DESTINATION bin)
